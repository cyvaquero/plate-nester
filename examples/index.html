import { nest, downloadBlob, plateFilename } from "../src/lib";

const ctrl = new AbortController();
const result = await nest(
  [{ name: "star.svg", svg: starSource, qty: 4 }],
  {
    mode: "shape", plateW: 300, plateH: 300, kerf: 0.1, gap: 1, margin: 3, rotStep: 90,
    budgetMs: 3000, signal: ctrl.signal,
    onLayout: (r) => console.log(r.stats.plates, "plates so far"),
  },
);
result.plates().forEach((svg, i) =>
  downloadBlob(new Blob([svg], { type: "image/svg+xml" }), plateFilename(i, result.stats.plates)));
